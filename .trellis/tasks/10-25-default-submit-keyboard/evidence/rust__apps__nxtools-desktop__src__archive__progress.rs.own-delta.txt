+                if !activation {
+                    down.set(false);
+                    return;
+                }
+                if event.prefer_character_input || event.keystroke.modifiers.modified() {
+                    down.set(false);

+                }
+                window.prevent_default();
+                cx.stop_propagation();
+            })
+            .capture_key_up(move |event, window, cx| {
